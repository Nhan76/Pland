"use client";

import React, { useEffect, useState } from "react";
import { motion, AnimatePresence } from "framer-motion";
import { ShieldAlert, Target, Globe, Activity, TrendingDown, Check, Loader2, Bot, MousePointer2, Sparkles } from "lucide-react";
import {
  AreaChart,
  Area,
  ResponsiveContainer,
  XAxis,
  YAxis,
  Tooltip
} from "recharts";

// Grid 1: AI Analytics Grid
const AIAnalysisGrid = () => {
  const [phase, setPhase] = useState(0);

  useEffect(() => {
    // Phase 0: Start (wait 1500)
    // Phase 1: Cursor clicks CTA (wait 400)
    // Phase 2: Slide up and fade out (wait 600)
    // Phase 3: Step 0 appears (wait 2000)
    // Phase 4: Step 1 appears (wait 2000)
    // Phase 5: Step 2 appears (wait 2000)
    // Phase 6: Summary appears (wait 5000)
    // Phase 7: Reset to 0 (wait 500)
    const delays = [1500, 400, 600, 2000, 2000, 2000, 5000, 500];
    const timer = setTimeout(() => {
      setPhase((p) => (p + 1) % 8);
    }, delays[phase] || 1000);
    return () => clearTimeout(timer);
  }, [phase]);

  const steps = [
    "Aggregating exchange balances...",
    "Analyzing cross-chain token correlations...",
    "Evaluating against defined risk guardrails..."
  ];

  return (
    <div className="relative flex h-[460px] flex-col overflow-hidden rounded-2xl border border-white/10 bg-bento-card backdrop-blur-sm p-6 sm:p-8 md:col-span-1">
      <div className="z-10 max-w-sm">
        <div className="mb-2 flex items-center space-x-2 text-white">
          <span className="text-xl font-semibold tracking-wider">AI Analytics</span>
        </div>
        <p className="text-white/60 text-sm xl:text-base leading-relaxed">
          Ask questions. Get answers grounded in your data. Summarize risks, check rules...
        </p>
      </div>

      <div className="relative flex flex-col w-full grow mt-6 z-10 justify-start">

        {/* Fake Cursor & CTA Group */}
        <div className="absolute inset-0 flex flex-col justify-center items-center pointer-events-none z-20 pb-[25%] lg:pb-16">
          <div className="relative flex justify-center">
            <motion.div
               animate={{ 
                 opacity: phase >= 2 ? 0 : 1, 
                 y: phase >= 2 ? -60 : -30, 
                 scale: phase === 1 ? 0.95 : 1 
               }}
               transition={{ duration: phase >= 2 ? 0.6 : 0.2 }}
               className="rounded-lg bg-primary px-8 py-3 text-sm font-bold text-white shadow-lg origin-center"
            >
               Analyze
            </motion.div>

            {/* Simulated Cursor */}
            <motion.div
               initial={{ x: 60, y: 60, opacity: 0 }}
               animate={{
                 x: phase === 0 ? 60 : (phase >= 1 && phase < 2 ? 10 : 10),
                 y: phase === 0 ? 60 : (phase >= 1 && phase < 2 ? -20 : -80),
                 opacity: phase === 0 ? 0 : (phase >= 2 ? 0 : 1),
                 scale: phase === 1 ? 0.9 : 1
               }}
               transition={{ 
                 duration: phase === 0 ? 0 : (phase === 1 ? 0.4 : 0.6), 
                 ease: "easeOut" 
               }}
               className="absolute right-0 bottom-0 z-50 text-white drop-shadow-lg"
            >
              <MousePointer2 className="h-6 w-6 fill-white stroke-black stroke-[1.5]" />
            </motion.div>
          </div>
        </div>

        {/* Analytics Steps Setup */}
        <div className="flex flex-col space-y-3 relative w-full pt-4">
          {steps.map((text, index) => {
            const stepIndex = index + 3; // Steps start at phase 3
            const isCompleted = phase > stepIndex;
            const isVisible = phase >= stepIndex;

            return (
              <motion.div
                key={index}
                layout
                initial={{ opacity: 0, y: 10 }}
                animate={{ opacity: isVisible ? 1 : 0, y: isVisible ? 0 : 10 }} 
                className={`flex items-start space-x-3 pointer-events-none`}    
              >
                <div className="mt-0.5">
                  {isCompleted && phase !== 7 ? (
                    <Check className="h-[18px] w-[18px] text-primary flex-shrink-0" />
                  ) : (
                    <Loader2 className={`h-[18px] w-[18px] flex-shrink-0 ${isVisible && !isCompleted ? 'text-primary animate-spin' : 'text-transparent'}`} />
                  )}
                </div>
                <span
                   className={`text-[14px] leading-tight font-medium tracking-wide ${
                    isCompleted && phase !== 7
                      ? "text-white/50 line-through"
                      : isVisible
                      ? "text-primary"
                      : "text-transparent"
                  }`}
                >
                  {text}
                </span>
              </motion.div>
            );
          })}

          {/* Final summary */}
          <motion.div
             layout
             initial={{ opacity: 0, y: 10 }}
             animate={{ opacity: phase === 6 ? 1 : 0, y: phase === 6 ? 0 : 10 }}
             transition={{ delay: 0.2 }}
             className="relative mt-4 wrap-break-word pointer-events-none"
          >
             <p className={`text-sm leading-relaxed ${phase === 6 ? 'text-white/80' : 'text-transparent'}`}>
               Portfolio is optimally diversified. <br className="hidden sm:block" /><span className={`${phase === 6 ? 'text-primary' : 'text-transparent'} font-medium`}>0 critical risk rules violated.</span> BTC dominance holds stable at 45%.
             </p>
          </motion.div>
        </div>
      </div>
    </div>
  );
};

// Grid 2: Risk Violation Alert Grid
const INITIAL_ALERTS = [
  { id: 1, text: "Unusual outbound transfer detected", type: "warning" },
  { id: 2, text: "Smart contract upgrade on holdings", type: "warning" },
  { id: 3, text: "Position size > 15%", type: "critical" },
  { id: 4, text: "High volatility detected", type: "warning" },
  { id: 5, text: "Drawdown threshold breached", type: "critical" },
];

const RiskAlertGrid = () => {
  const [alerts, setAlerts] = useState(INITIAL_ALERTS);

  useEffect(() => {
    let count = 5;
    const interval = setInterval(() => {
      count++;
      
      const newAlert = {
        id: count,
        text: count % 2 === 0 ? "Drawdown threshold breached" : "High volatility detected",
        type: count % 2 === 0 ? "critical" : "warning",
      };

      setAlerts((prev) => [...prev.slice(1), newAlert]); 
    }, 3000); // 3s interval for active scrolling effect

    return () => clearInterval(interval);
  }, []);

  return (
    <div className="relative flex h-[460px] flex-col overflow-hidden rounded-2xl border border-white/10 bg-bento-card backdrop-blur-sm p-6 sm:p-8 md:col-span-2">
      <div className="z-10">
        <div className="mb-2 flex items-center space-x-2 text-white">
          <span className="text-xl font-semibold tracking-wider">Live Alerts</span>
        </div>
        <p className="text-white/60 text-base leading-relaxed">
          Real-time monitoring of on-chain activities, volatility spikes, and portfolio parameter breaches.
        </p>
      </div>

      <div className="relative flex flex-grow flex-col justify-center space-y-3 pb-2 overflow-hidden">
        <AnimatePresence mode="popLayout">
          {alerts.map((alert, index) => {
            const centerIndex = 2; // Middle index for 5 items
            const dist = Math.abs(centerIndex - index);

            return (
              <motion.div
                layout
                key={alert.id}
                initial={{ opacity: 0, y: 20, scale: 0.8 }}
                animate={{ 
                  opacity: 1 - dist * 0.3, 
                  scale: 1 - dist * 0.05, 
                  y: 0 
                }}
                exit={{ opacity: 0, y: -40, scale: 0.8 }}
                transition={{ duration: 0.6, ease: "easeInOut" }}
                className={`flex items-start space-x-3 rounded-lg border px-4 py-3 backdrop-blur-md ${
                  alert.type === "critical" 
                    ? "border-red-500/30 bg-red-500/10 text-red-500" 
                    : "border-orange-500/30 bg-orange-500/10 text-orange-500"
                }`}
                style={{ zIndex: 10 - dist }}
              >
              {alert.type === "critical" ? (
                <ShieldAlert className="mt-0.5 h-5 w-5 flex-shrink-0 animate-pulse" />
              ) : (
                <TrendingDown className="mt-0.5 h-5 w-5 flex-shrink-0" />
              )}
              <div>
                <p className="text-sm font-semibold">Rule Violated</p>
                <p className={`text-xs ${alert.type === "critical" ? "text-red-400" : "text-orange-400"}`}>
                  {alert.text}
                </p>
              </div>
            </motion.div>
          );
        })}
        </AnimatePresence>
      </div>
    </div>
  );
};
